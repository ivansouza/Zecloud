Tradução do Shader "Cauldron"

Código Original (Code-Golf) ~215 bytes 
f z, d
@(40)
{
f3 p = z * nor(2*C.rgb - R.xyy)
p.zy+=3
d=2; @(7) d/=.8,
p +=.6*p.y*cos(p.yzx * d - f3(T,,)*6) / d
z += d = .02 + abs(len(p.xz) + p.y*.4 -1) / 9
O += (sin(f4(5,0,2,)-p.y) + 2) / d
}
O = tanh(O*O / 15e5)

Como funciona a matemática:

  • @(40) executa 40 passos de raymarching volumétrico ao longo do raio.
  • p.zy += 3 desloca a câmera para trás e para cima para enquadrar o caldeirão.
  • @(7) d/=.8 aplica 7 oitavas de distorção harmônica (cos(p.yzx * d)) escalada por p.y.
  • abs(len(p.xz) + p.y*.4 - 1) define um cone oco translúcido cujas paredes acumulam luz.
  • tanh(O*O / 15e5) realiza o mapeamento de tom (tone mapping) suave evitando estouro de branco.
WebGL 2.0 (GLSL ES 3.00) 
#version 300 es
precision highp float;

uniform vec2 u_resolution;
uniform float u_time;
uniform vec4 u_Color;

void main() {
    float z = 0.0, d = 0.0;
    vec4 O = vec4(0.0);
    
    // Direção normalizada do raio: nor(2*C.rgb - R.xyy)
    vec3 rd = normalize(vec3(2.0 * gl_FragCoord.xy - u_resolution.xy, -u_resolution.y));

    for (int i = 0; i < 40; i++) {
        vec3 p = z * rd;
        p.zy += 3.0;

        // 7 oitavas de turbulência fluida
        d = 2.0;
        for (int j = 0; j < 7; j++) {
            d /= 0.8;
            p += 0.6 * p.y * cos(p.yzx * d - vec3(u_time * 6.0, 0.0, 0.0)) / d;
        }

        // Distância estimada para o cone volumétrico
        z += d = 0.02 + abs(length(p.xz) + p.y * 0.4 - 1.0) / 9.0;

        // Acumulação cromática baseada na altura p.y
        O += (sin(vec4(5.0, 0.0, 2.0, 0.0) - p.y) + 2.0) / d;
    }

    // Tone mapping hiperbólico
    u_Color = tanh(O * O / 15e5);
    u_Color.a = 1.0;
}
